<read_first>

- src/services/wallpaper.service.ts
- src/config/constants.ts
</read_first>

<action>
修改 `src/stores/modules/wallpaper/index.ts`：

1. 添加 LRUCache 导入：
```typescript
import LRUCache from 'lru-cache'
import { CACHE_CONFIG } from '@/config/constants'
```

2. 定义 CacheItem 接口：
```typescript
interface SearchCacheItem {
  data: unknown
  timestamp: number
}
```

3. 创建 LRUCache 实例：
```typescript
/** 搜索结果缓存 */
const searchCache = new LRUCache<string, SearchCacheItem>({
  maxSize: CACHE_CONFIG.SEARCH_MAX_SIZE_BYTES,
  ttl: CACHE_CONFIG.SEARCH_TTL_MS,
  sizeCalculation: (value) => JSON.stringify(value.data).length,
})
```

4. 添加缓存操作方法：
```typescript
function getCachedSearch<T>(key: string): T | null {
  const item = searchCache.get(key)
  return item ? (item.data as T) : null
}

function setCachedSearch(key: string, data: unknown): void {
  searchCache.set(key, { data, timestamp: Date.now() })
}

function clearSearchCache(): void {
  searchCache.clear()
}

function generateCacheKey(url: string, params?: unknown): string {
  return `${url}:${JSON.stringify(params || {})}`
}

function getSearchCacheStats(): { size: number; hits: number; misses: number } {
  return {
    size: searchCache.size,
    hits: 0,
    misses: 0,
  }
}
```

5. 在 return 中暴露新方法
</action>

- useWallpaperStore 包含 searchCache (LRUCache)
- Store 包含 getCachedSearch、setCachedSearch、clearSearchCache 方法
- TypeScript 编译通过
</acceptance_criteria>

2. 修改 search 方法，使用 FavoritesStore 的缓存：
```typescript
async search(params: GetParams | null, options?: { signal?: AbortSignal }): Promise<IpcResponse<WallpaperSearchResult>> {
  // ...existing code until favorite status injection
  
  if (result.success && result.data) {
    if (result.data.data.length > 0) {
      const wallpaperIds = result.data.data.map((item) => item.id)
      
      // 使用 Store 的缓存
      const { useFavoritesStore } = await import('@/stores')
      const favoritesStore = useFavoritesStore()
      
      // 获取已缓存的和未缓存的 ID
      const cachedStatus: Record<string, 0 | 1 | 2> = {}
      const uncachedIds: string[] = []
      
      for (const id of wallpaperIds) {
        const cached = favoritesStore.getFavoriteStatus(id)
        if (cached !== undefined) {
          cachedStatus[id] = cached
        } else {
          uncachedIds.push(id)
        }
      }
      
      // 加载未缓存的
      if (uncachedIds.length > 0) {
        await favoritesStore.loadFavoriteStatusMap(uncachedIds)
      }
      
      // 构建最终状态映射
      const statusMap: Record<string, 0 | 1 | 2> = { ...cachedStatus }
      for (const id of wallpaperIds) {
        const status = favoritesStore.getFavoriteStatus(id)
        if (status !== undefined) {
          statusMap[id] = status
        }
      }
      
      result.data.data = result.data.data.map((item) => ({
        ...item,
        is_favorite: statusMap[item.id] ?? 0,
      }))
    }
    
    this.setCache(cacheKey, result.data)
  }
  
  return result
}
```

3. 删除 clearFavoriteStatusCache 方法（已迁移到 Store）
</action>

```typescript
async search(params: GetParams | null, options?: { signal?: AbortSignal }): Promise<IpcResponse<WallpaperSearchResult>> {
  const { useWallpaperStore } = await import('@/stores')
  const wallpaperStore = useWallpaperStore()
  
  // 过滤空值参数
  const filteredParams: Record<string, unknown> = {}
  if (params) {
    Object.entries(params).forEach(([key, value]) => {
      if (value !== null && value !== undefined && value !== '') {
        filteredParams[key] = value
      }
    })
  }
  
  // 使用 Store 缓存
  const cacheKey = wallpaperStore.generateCacheKey('/search', filteredParams)
  const cachedData = wallpaperStore.getCachedSearch<WallpaperSearchResult>(cacheKey)
  if (cachedData) {
    return { success: true, data: cachedData }
  }
  
  // ...existing API call logic
  
  // 成功时设置 Store 缓存
  if (result.success && result.data) {
    wallpaperStore.setCachedSearch(cacheKey, result.data)
  }
  
  return result
}
```
