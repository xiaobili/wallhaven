```typescript
/**
 * 下载任务仓储
 * 封装下载任务相关的 IPC 调用
 */

import type {
  IpcResponse,
  PendingDownload,
  ResumeDownloadParams,
} from '@/types/ipc'
import { electronClient } from '@/clients'

/**
 * 下载任务仓储
 */
export const downloadTaskRepository = {
  /**
   * 选择文件夹
   */
  async selectFolder(): Promise<IpcResponse<string>> {
    return electronClient.selectFolder()
  },

  /**
   * 检查文件是否存在
   */
  async fileExists(path: string): Promise<IpcResponse<boolean>> {
    return electronClient.fileExists(path)
  },

  /**
   * 下载壁纸（同步模式）
   */
  async downloadWallpaper(params: {
    url: string
    filename: string
    saveDir: string
  }): Promise<IpcResponse<string>> {
    return electronClient.downloadWallpaper(params)
  },

  /**
   * 启动下载任务
   */
  async startDownloadTask(params: {
    taskId: string
    url: string
    filename: string
    saveDir: string
  }): Promise<IpcResponse<string>> {
    return electronClient.startDownloadTask(params)
  },

  /**
   * 暂停下载任务
   */
  async pauseDownloadTask(taskId: string): Promise<IpcResponse<void>> {
    return electronClient.pauseDownloadTask(taskId)
  },

  /**
   * 取消下载任务
   */
  async cancelDownloadTask(taskId: string): Promise<IpcResponse<void>> {
    return electronClient.cancelDownloadTask(taskId)
  },

  /**
   * 恢复下载任务
   */
  async resumeDownloadTask(params: ResumeDownloadParams): Promise<IpcResponse<string>> {
    return electronClient.resumeDownloadTask(params)
  },

  /**
   * 获取待恢复的下载任务列表
   */
  async getPendingDownloads(): Promise<IpcResponse<PendingDownload[]>> {
    return electronClient.getPendingDownloads()
  },

  /**
   * 清理孤儿临时文件
   */
  async cleanupOrphanFiles(saveDir: string): Promise<
    IpcResponse<{ filesDeleted: number; stateFilesDeleted: number }>
  > {
    return electronClient.cleanupOrphanFiles(saveDir)
  },

  /**
   * 注册下载进度监听器
   */
  onDownloadProgress(callback: (data: unknown) => void): void {
    electronClient.onDownloadProgress(callback)
  },
}
```
</action>
